cmake_minimum_required(VERSION 3.16)
project(borno-fcitx5 VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Fcitx5Core 5.1.8 REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
include(CTest)
set(BORNOPATA_SOURCES src/bornopata.cpp src/ipc/connection.cpp src/key/mapping.cpp src/popup/bridge.cpp src/popup/channel.cpp)

# The popup surface needs the Wayland client library, the protocol XML of wayland-protocols and the Fcitx5 Wayland module;
# without any of them the plugin is built with a presenter that is never available.
find_package(PkgConfig)
find_package(Fcitx5ModuleWayland QUIET)
set(BORNOPATA_POPUP_POSSIBLE OFF)
if(PkgConfig_FOUND AND Fcitx5ModuleWayland_FOUND)
    pkg_check_modules(WAYLAND_CLIENT QUIET IMPORTED_TARGET wayland-client)
    pkg_get_variable(WAYLAND_PROTOCOLS_DIR wayland-protocols pkgdatadir)
    find_program(WAYLAND_SCANNER wayland-scanner)
    if(WAYLAND_CLIENT_FOUND AND WAYLAND_PROTOCOLS_DIR AND WAYLAND_SCANNER)
        set(BORNOPATA_POPUP_POSSIBLE ON)
    endif()
endif()
option(BORNOPATA_WAYLAND_POPUP "Show the candidate window in the input-method popup surface" ${BORNOPATA_POPUP_POSSIBLE})
if(BORNOPATA_WAYLAND_POPUP)
    set(protocol_dir ${CMAKE_CURRENT_BINARY_DIR}/protocol)
    file(MAKE_DIRECTORY ${protocol_dir})
    set(protocol_sources)
    foreach(xml
            ${CMAKE_CURRENT_SOURCE_DIR}/protocol/input-method-unstable-v2.xml
            ${WAYLAND_PROTOCOLS_DIR}/stable/viewporter/viewporter.xml
            ${WAYLAND_PROTOCOLS_DIR}/staging/fractional-scale/fractional-scale-v1.xml)
        get_filename_component(stem ${xml} NAME_WE)
        add_custom_command(OUTPUT ${protocol_dir}/${stem}-client-protocol.h
            COMMAND ${WAYLAND_SCANNER} client-header ${xml} ${protocol_dir}/${stem}-client-protocol.h
            DEPENDS ${xml})
        add_custom_command(OUTPUT ${protocol_dir}/${stem}-protocol.c
            COMMAND ${WAYLAND_SCANNER} private-code ${xml} ${protocol_dir}/${stem}-protocol.c
            DEPENDS ${xml})
        list(APPEND protocol_sources ${protocol_dir}/${stem}-client-protocol.h ${protocol_dir}/${stem}-protocol.c)
    endforeach()
    set_source_files_properties(${protocol_sources} PROPERTIES GENERATED TRUE)
    list(APPEND BORNOPATA_SOURCES src/popup/wayland/display.cpp src/popup/wayland/presenter.cpp src/popup/wayland/proxy.cpp
        src/popup/wayland/surface.cpp ${protocol_sources})
else()
    list(APPEND BORNOPATA_SOURCES src/popup/wayland/none.cpp)
endif()
function(bornopata_link target)
    target_include_directories(${target} PRIVATE src)
    if(BORNOPATA_WAYLAND_POPUP)
        target_include_directories(${target} PRIVATE src/popup/wayland ${protocol_dir})
        target_link_libraries(${target} PRIVATE PkgConfig::WAYLAND_CLIENT Fcitx5::Module::Wayland)
    endif()
endfunction()
add_library(bornopata MODULE src/factory.cpp ${BORNOPATA_SOURCES})
bornopata_link(bornopata)
target_link_libraries(bornopata PRIVATE Fcitx5::Core nlohmann_json::nlohmann_json)
set_target_properties(bornopata PROPERTIES PREFIX "")
install(TARGETS bornopata LIBRARY DESTINATION lib/fcitx5)
install(FILES data/addon/bornopata.conf DESTINATION share/fcitx5/addon)
install(FILES data/inputmethod/bornopata.conf DESTINATION share/fcitx5/inputmethod)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    pkg_check_modules(SYSTEMD REQUIRED IMPORTED_TARGET libsystemd)
    set(plugin_tests integration lifecycle real realpopup reentry privacy transport popup)
    if(BORNOPATA_WAYLAND_POPUP)
        list(APPEND plugin_tests proxy guard)
    endif()
    foreach(test ${plugin_tests})
        add_executable(test-${test} tests/${test}.cpp ${BORNOPATA_SOURCES})
        target_include_directories(test-${test} PRIVATE tests)
        bornopata_link(test-${test})
        target_link_libraries(test-${test} PRIVATE Fcitx5::Core nlohmann_json::nlohmann_json Threads::Threads PkgConfig::SYSTEMD)
        target_compile_options(test-${test} PRIVATE -UNDEBUG)
    endforeach()
    set(BORNOPATA_SERVER_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/../../../target/debug/borno-linux-server" CACHE FILEPATH "The built Rust Server")
    foreach(case old-version timeout malformed session)
        add_test(NAME transport-${case} COMMAND test-transport ${case})
        set_tests_properties(transport-${case} PROPERTIES TIMEOUT 5)
    endforeach()
    add_test(NAME real-popup COMMAND test-realpopup "${BORNOPATA_SERVER_EXECUTABLE}")
    set_tests_properties(real-popup PROPERTIES TIMEOUT 25)
    add_test(NAME real-server COMMAND test-real "${BORNOPATA_SERVER_EXECUTABLE}")
    set_tests_properties(real-server PROPERTIES TIMEOUT 15)
    foreach(case password sensitive disable reset ui-reset focus click destroy display-failure)
        add_test(NAME reentry-${case} COMMAND test-reentry "${BORNOPATA_SERVER_EXECUTABLE}" ${case})
        set_tests_properties(reentry-${case} PROPERTIES TIMEOUT 15)
    endforeach()
    foreach(case sensitive-reset sensitive-focus sensitive-key password-reset disable-focus superseded-password superseded-disable superseded-sensitive)
        add_test(NAME privacy-${case} COMMAND test-privacy "${BORNOPATA_SERVER_EXECUTABLE}" ${case})
        set_tests_properties(privacy-${case} PROPERTIES TIMEOUT 15)
    endforeach()
    if(BORNOPATA_WAYLAND_POPUP)
        add_test(NAME proxy-finder COMMAND test-proxy)
        add_test(NAME surface-guard COMMAND test-guard)
    endif()
    foreach(scenario show release stale hangup unavailable)
        add_test(NAME popup-${scenario} COMMAND test-popup ${scenario})
        set_tests_properties(popup-${scenario} PROPERTIES TIMEOUT 10)
    endforeach()
    foreach(mode both native failure mismatch)
        add_test(NAME panel-${mode} COMMAND test-integration ${mode})
    endforeach()
    foreach(case focusout-panel focusout-client-owned focusout-inline capability-deactivate password sensitive disabled empty shift-empty)
        add_test(NAME lifecycle-${case} COMMAND test-lifecycle ${case})
        set_tests_properties(lifecycle-${case} PROPERTIES TIMEOUT 10)
    endforeach()
endif()
