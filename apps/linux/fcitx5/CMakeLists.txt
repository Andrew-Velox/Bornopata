cmake_minimum_required(VERSION 3.16)
project(borno-fcitx5 VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Fcitx5Core 5.1.8 REQUIRED)
find_package(nlohmann_json 3.2 REQUIRED)
include(CTest)
set(BORNOPATA_SOURCES src/bornopata.cpp src/ipc/connection.cpp src/key/mapping.cpp)
add_library(bornopata MODULE src/factory.cpp ${BORNOPATA_SOURCES})
target_include_directories(bornopata PRIVATE src)
target_link_libraries(bornopata PRIVATE Fcitx5::Core nlohmann_json::nlohmann_json)
set_target_properties(bornopata PROPERTIES PREFIX "")
install(TARGETS bornopata LIBRARY DESTINATION lib/fcitx5)
install(FILES data/addon/bornopata.conf DESTINATION share/fcitx5/addon)
install(FILES data/inputmethod/bornopata.conf DESTINATION share/fcitx5/inputmethod)
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    foreach(test integration lifecycle real reentry privacy transport)
        add_executable(test-${test} tests/${test}.cpp ${BORNOPATA_SOURCES})
        target_include_directories(test-${test} PRIVATE src tests)
        target_link_libraries(test-${test} PRIVATE Fcitx5::Core nlohmann_json::nlohmann_json Threads::Threads)
        target_compile_options(test-${test} PRIVATE -UNDEBUG)
    endforeach()
    set(BORNOPATA_SERVER_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/../../../target/debug/borno-linux-server" CACHE FILEPATH "The built Rust Server")
    foreach(case old-version timeout malformed session)
        add_test(NAME transport-${case} COMMAND test-transport ${case})
        set_tests_properties(transport-${case} PROPERTIES TIMEOUT 5)
    endforeach()
    add_test(NAME real-server COMMAND test-real "${BORNOPATA_SERVER_EXECUTABLE}")
    set_tests_properties(real-server PROPERTIES TIMEOUT 15)
    foreach(case password sensitive disable reset ui-reset focus click destroy display-failure)
        add_test(NAME reentry-${case} COMMAND test-reentry "${BORNOPATA_SERVER_EXECUTABLE}" ${case})
        set_tests_properties(reentry-${case} PROPERTIES TIMEOUT 15)
    endforeach()
    foreach(case sensitive-reset sensitive-focus sensitive-key password-reset disable-focus superseded-password superseded-disable superseded-sensitive)
        add_test(NAME privacy-${case} COMMAND test-privacy "${BORNOPATA_SERVER_EXECUTABLE}" ${case})
        set_tests_properties(privacy-${case} PROPERTIES TIMEOUT 15)
    endforeach()
    foreach(mode both native failure mismatch)
        add_test(NAME panel-${mode} COMMAND test-integration ${mode})
    endforeach()
    foreach(case focusout-panel focusout-client-owned focusout-inline capability-deactivate password sensitive disabled empty shift-empty)
        add_test(NAME lifecycle-${case} COMMAND test-lifecycle ${case})
        set_tests_properties(lifecycle-${case} PROPERTIES TIMEOUT 10)
    endforeach()
endif()
